"""
transport_profiles.py

Surface-flow profiles for the SFT model: meridional circulation and
differential rotation.

Sign convention (read this before worrying about a "wrong" sign)
----------------------------------------------------------------
Colatitude ``theta`` runs 0 at the north pole to ``pi`` at the south pole, so
the ``+theta`` direction points *southward everywhere*.  The solver advects
the radial field with the **colatitude component of the velocity**,
``u_theta`` -- that is what :class:`sft2d.src.operators.Advection` consumes and
what :func:`meridional_flow` returns.

For a normal solar **poleward** meridional flow (positive ``peak_speed``):

    * northern hemisphere: poleward = toward theta=0  => u_theta < 0
    * southern hemisphere: poleward = toward theta=pi => u_theta > 0

So ``meridional_flow`` is **negative in the north and positive in the south**
for a poleward flow.  That looks upside-down only because it is the
*colatitude* velocity.  Expressed as the more intuitive **latitude velocity**
``v_lat`` (northward positive, ``v_lat = -u_theta``) the same poleward flow is
positive in the north and negative in the south -- which is what you expect
from meridional-flow figures in the literature.  Use
:func:`meridional_flow_latitude` when you want to *plot* the flow; use
:func:`meridional_flow` to *drive the model*.

Verified: a flux blob advected with ``meridional_flow(grid, +15)`` migrates
toward the nearest pole in both hemispheres (poleward).  A **negative**
``peak_speed`` reverses this into an unphysical equatorward flow, which will
prevent trailing-polarity flux from ever reaching the poles -- do not use a
negative ``peak_speed`` expecting to "fix" a polar-reversal problem; that is a
flux-amplitude / tilt question, not a flow-direction one.
"""

def meridional_flow(grid, peak_speed=15.0, profile="yeates2020"):
    """Poleward meridional-circulation profile as the colatitude velocity ``u_theta``.

    Parameters
    ----------
    grid : dict
        Grid from :func:`sft2d.src.grid.create_grid`.
    peak_speed : float
        Peak poleward speed [m/s].  **Positive = poleward** (the physical case).
    profile : {'yeates2020', 'schuessler-baumann'}
        ``'yeates2020'`` (default): ``vs`` profile, peaks near +/-45 deg.
        ``'schuessler-baumann'``: ``-sin(2 lat) exp(pi(1 - 2|lat|/pi))``, the
        profile used in Yeates's ``sft_data``; peaks nearer +/-35 deg and falls
        off faster toward the poles (typical peak speed ~11 m/s).

    Returns
    -------
    ndarray, shape (n_theta, n_phi)
        ``u_theta`` [m/s]: negative in the north, positive in the south for a
        poleward flow.  Feed this straight into
        :class:`sft2d.src.operators.Advection` / :func:`sft2d.src.stepper.evolve`.
    """
    theta = grid["colatitude"]
    n_phi = grid["longitude"].shape[0]
    lat = 0.5 * np.pi - theta

    if profile in ("yeates2020", "default"):
        # vs() is defined on latitude; the latitude here is (pi/2 - theta).  The
        # extra sign makes the result the *colatitude* velocity u_theta (poleward
        # for positive peak_speed), which is what the advection operator expects.
        v_theta_1d = vs(theta - np.pi / 2, v0=peak_speed)
    elif profile in ("schuessler-baumann", "sb", "schussler-baumann"):
        f = -np.sin(2.0 * lat) * np.exp(np.pi * (1.0 - 2.0 * np.abs(lat) / np.pi))
        v_theta_1d = peak_speed * f / np.max(np.abs(f))
    else:
        raise ValueError("profile must be 'yeates2020' or 'schuessler-baumann'")

    return np.tile(v_theta_1d, (n_phi, 1)).T

def meridional_flow_latitude(grid, peak_speed=15.0):
    """Same flow as :func:`meridional_flow`, but as the **latitude velocity** ``v_lat``.

    ``v_lat = -u_theta`` is northward-positive, so a poleward flow is positive
    in the north and negative in the south -- the convention used in most
    meridional-flow figures.  This is a *plotting / inspection* helper; the
    solver is driven with :func:`meridional_flow`.
    """
    return -meridional_flow(grid, peak_speed=peak_speed)

def differential_rotation(grid, rotation="solar", frame="carrington"):
    """Differential-rotation angular velocity ``Omega(theta)`` [rad/s].

    Parameters
    ----------
    grid : dict
        Grid from :func:`sft2d.src.grid.create_grid`.
    rotation : {'solar', 'rigid'}
        ``'solar'`` uses the Snodgrass-type profile; ``'rigid'`` a 2-day
        solid-body rotation (a test case).
    frame : {'carrington', 'synodic'}
        Reference frame for the solar profile.

    Returns
    -------
    ndarray, shape (n_theta, n_phi)
        ``Omega`` [rad/s].
    """
    theta = grid["colatitude"]
    n_phi = grid["longitude"].shape[0]
    Colatitude = np.tile(theta, (n_phi, 1)).T

    # degrees/day -> radians/second
    rotation_rate_fact = 2.0201e-7

    if rotation == "solar":
        if frame == "carrington":
            base = 13.38 - 360.0 / 27.2753
        elif frame == "synodic":
            base = 13.38
        else:
            raise ValueError("For solar rotation, frame must be 'carrington' or 'synodic'.")
        omega_diff = (
            base
            - 2.30 * np.cos(Colatitude) ** 2
            - 1.62 * np.cos(Colatitude) ** 4
        ) * rotation_rate_fact
    elif rotation == "rigid":
        rotation_period_days = 2.0
        omega_diff = np.full(Colatitude.shape, 360.0 / rotation_period_days)
    else:
        raise ValueError("rotation must be 'solar' or 'rigid'.")

    return omega_diff




def vs(lat, v0=1.0, p=2.33):
    """Yeates (2020, Sol. Phys.) meridional-flow shape as a function of latitude.

    ``v(lat) = Du * sin(lat) * cos(lat)^p`` with ``Du`` chosen so the peak speed
    equals ``v0``.  Positive ``v0`` gives a poleward (northward in the north)
    latitude velocity; :func:`meridional_flow` converts it to ``u_theta``.
    """
    Du = v0 * (1.0 + p) ** (0.5 * (p + 1.0)) / p ** (0.5 * p)
    return Du * np.sin(lat) * (np.cos(lat)) ** p


def active_region_inflow(grid, field, amplitude=1.0, smooth_deg=6.0,
                         max_speed=None):
    """Converging flow toward active latitudes, as a colatitude velocity ``u_theta``.

    Observed surface flows converge on activity belts by a few m/s, and the
    convergence scales with how much flux is present.  In an SFT model this acts
    as a brake: flux that would otherwise be carried poleward is held back near
    the activity latitudes, and the brake is strongest in the strongest cycles.

    That behaviour cannot be produced by any static flow, which is why it matters
    here.  Re-scoring a 686-member sweep over flow profile, ``du``, ``eta`` and
    ``tau`` showed the model over-responds in cycle 25 relative to cycle 24 for
    *every* combination (523/523, median 2.7x): a parameter that is constant in
    time cannot behave differently in two cycles.

    The flow is taken proportional to the latitudinal gradient of the smoothed,
    longitude-averaged unsigned field, converging on concentrations::

        v_lat = +amplitude * d<|B|>/dlambda

    so it vanishes when the surface is quiet and grows with activity by itself,
    with no explicit cycle dependence.

    Parameters
    ----------
    grid : dict
        Grid from :func:`sft2d.src.grid.create_grid`.
    field : ndarray
        Current radial field ``(n_theta, n_phi)``.
    amplitude : float
        Scale factor in m/s per (G/radian).  0 disables the term.  Calibrated on
        a real model state -- the busiest epoch of a SHARPS-driven run, zonal-mean
        |B| peaking at 64 G near 16 deg -- ``amplitude = 0.057`` gives a 3 m/s peak
        inflow and ``0.076`` gives 4 m/s, the observed range.  The value depends
        on ``smooth_deg``, since both enter through the same gradient, so the two
        must be quoted together.
    smooth_deg : float
        Gaussian smoothing applied to the zonal-mean profile before
        differentiating, in degrees.  Without it the gradient is dominated by
        individual regions rather than by the belt.  It must stay well below the
        separation of the two activity belts: at 15 deg the northern and southern
        belts (about 30 deg apart) merge into a single hump centred on the
        equator, the gradient stops changing sign at the belt latitude, and the
        flow diverges from the belts instead of converging on them.  6 deg keeps
        them resolved.
    max_speed : float, optional
        Clip the inflow to this speed [m/s].  A safety rail: an unclipped
        gradient term can spike where a large region has just emerged, and the
        advective time step then collapses.

    Returns
    -------
    ndarray
        ``u_theta`` perturbation [m/s], same shape as ``field``, to be **added**
        to the background meridional flow.
    """
    if amplitude == 0.0:
        return np.zeros_like(field)

    theta = grid["colatitude"]
    lat = 0.5 * np.pi - theta
    prof = np.mean(np.abs(field), axis=1)          # zonal-mean unsigned field

    # Gaussian smoothing in latitude, on the model's own (non-uniform) mesh.
    sig = np.deg2rad(smooth_deg)
    d = lat[:, None] - lat[None, :]
    w = np.exp(-0.5 * (d / sig) ** 2)
    w /= w.sum(axis=1, keepdims=True)
    sm = w @ prof

    # Converging on maxima means moving UP the gradient: below the belt the flow
    # is poleward (d|B|/dlat > 0), above it equatorward (d|B|/dlat < 0), so the
    # sign is positive. Getting this backwards produces a flow that DIVERGES from
    # the activity belts and accelerates flux poleward -- the opposite of the
    # intended brake, and it would have made the polar over-build worse.
    v_lat = amplitude * np.gradient(sm, lat)
    if max_speed is not None:
        v_lat = np.clip(v_lat, -abs(max_speed), abs(max_speed))

    # u_theta = -v_lat (see the sign note at the top of this module)
    return np.tile(-v_lat, (grid["longitude"].shape[0], 1)).T


def active_region_inflow_2d(grid, field, amplitude=0.027, smooth_deg=6.0,
                            max_speed=8.0):
    """Local converging flow around each active region, as ``(u_theta, d_omega)``.

    The zonal-mean version, :func:`active_region_inflow`, spreads the brake over a
    whole latitude band.  Real inflows are local: they appear around an individual
    region when it emerges, weaken as it decays, and travel with it.  Representing
    that faithfully sounds like it needs the regions tracked individually, with a
    flow parcel attached to each centroid.

    It does not, because the field already carries that information.  Taking the
    gradient of the *2-D* smoothed unsigned field gives a flow that converges on
    every local concentration wherever it happens to be, strengthens when a large
    region emerges, follows it as differential rotation and meridional flow carry
    it, and fades as diffusion erases it -- with no region list, no centroids and
    no matching between one day and the next.  The tracking is implicit in the
    field.

    Both components are returned, because a local feature converges from all
    directions, not just in latitude:

    * ``u_theta``  [m/s], to be added to the meridional flow;
    * ``d_omega``  [rad/s], to be added to the differential rotation.

    Parameters
    ----------
    amplitude : float
        m/s per (G/radian).  The 2-D gradient around a single region is much
        steeper than the zonal-mean one, so this is NOT the same number as for
        :func:`active_region_inflow`: 0.027 here gives a ~4 m/s peak beside a
        strong (70 G) region, where the zonal-mean form needs 0.076.
    smooth_deg : float
        Gaussian width in degrees, applied in both latitude and longitude.  It
        sets the size of the inflow cell around a region.
    max_speed : float, optional
        Clip on each component [m/s].  A freshly emerged region has a very steep
        gradient, and an unclipped inflow there would dominate the advective time
        step.

    Overlapping and neighbouring regions
    ------------------------------------
    Because the flow is a gradient of the *total* smoothed field rather than a
    parcel attached to each region, nearby regions combine on their own:

    * beyond about 30 deg apart (roughly 5x ``smooth_deg``) each keeps its own
      inflow cell at single-region strength;
    * inside about 12 deg they merge into one cell centred on the complex, with a
      strength set by the combined flux;
    * exactly co-located, two regions give precisely the inflow of one region of
      their summed strength -- the response is linear in |B|, so a complex twice
      as strong drives twice the inflow.

    Between two separated regions the flow reverses through a stagnation point.
    That is a saddle, not an artefact: any flow converging on two centres must
    have one between them, and the speed there goes to zero (~0.01 m/s against
    ~2.7 m/s at the region edges).
    """
    if amplitude == 0.0:
        z = np.zeros_like(field)
        return z, z

    theta = grid["colatitude"]
    phi = grid["longitude"]
    nt, npm = field.shape
    sig = np.deg2rad(smooth_deg)

    # Separable Gaussian smoothing: a dense matrix in colatitude (the mesh is not
    # uniform) and a circular convolution in longitude (which is).
    d = theta[:, None] - theta[None, :]
    wt = np.exp(-0.5 * (d / sig) ** 2)
    wt /= wt.sum(axis=1, keepdims=True)
    sm = wt @ np.abs(field)

    dphi = phi[1] - phi[0]
    # The kernel must have lag zero at index 0 for a circular convolution, or the
    # result is rotated: a centred-then-rolled kernel shifted the whole field by
    # a quarter turn, which put every inflow cell 90 deg away from its region.
    k = np.arange(npm)
    lag = np.minimum(k, npm - k) * dphi
    wp = np.exp(-0.5 * (lag / sig) ** 2)
    wp /= wp.sum()
    sm = np.real(np.fft.ifft(np.fft.fft(sm, axis=1) * np.fft.fft(wp), axis=1))

    # Converge on maxima: move up the gradient (see active_region_inflow).
    # u_theta = -v_lat and d/dlat = -d/dtheta, so u_theta = +a * d(sm)/dtheta.
    u_theta = amplitude * np.gradient(sm, theta, axis=0)

    # Longitudinal component, per unit arc: (1/sin theta) d(sm)/dphi.
    st = np.sin(theta)[:, None]
    v_phi = amplitude * np.gradient(sm, phi, axis=1) / np.maximum(st, 1e-6)

    if max_speed is not None:
        u_theta = np.clip(u_theta, -abs(max_speed), abs(max_speed))
        v_phi = np.clip(v_phi, -abs(max_speed), abs(max_speed))

    # omega = v_phi / (R sin theta)
    d_omega = v_phi / np.maximum(R_SUN_M * st, 1.0)
    return u_theta, d_omega
